cmake_minimum_required(VERSION 3.13)

include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)

project(CO2_Control_System C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)

pico_sdk_init()

set(FREERTOS_KERNEL_PATH
${CMAKE_CURRENT_LIST_DIR}/lib/FreeRTOS-Kernel
)

add_executable(CO2-Control-System
src/main.cpp
src/app.cpp
src/modbus.cpp
src/sensors.cpp
src/controller.cpp
src/ui.cpp
src/network.cpp
src/storage.cpp

```
${FREERTOS_KERNEL_PATH}/portable/tasks.c
${FREERTOS_KERNEL_PATH}/portable/queue.c
${FREERTOS_KERNEL_PATH}/portable/list.c
${FREERTOS_KERNEL_PATH}/portable/timers.c
${FREERTOS_KERNEL_PATH}/portable/event_groups.c
${FREERTOS_KERNEL_PATH}/portable/stream_buffer.c

${FREERTOS_KERNEL_PATH}/portable/GCC/MemMang/heap_4.c

${FREERTOS_KERNEL_PATH}/portable/GCC/ARM_CM0/port.c
```

)

target_include_directories(CO2-Control-System PRIVATE
${CMAKE_CURRENT_LIST_DIR}/include
${FREERTOS_KERNEL_PATH}/include
${FREERTOS_KERNEL_PATH}/portable/GCC/ARM_CM0
)

target_link_libraries(CO2-Control-System
pico_stdlib
hardware_gpio
hardware_timer
hardware_uart
hardware_i2c
hardware_flash
hardware_sync
)

pico_enable_stdio_usb(CO2-Control-System 1)
pico_enable_stdio_uart(CO2-Control-System 0)

pico_add_bin_output(CO2-Control-System)
pico_add_hex_output(CO2-Control-System)
