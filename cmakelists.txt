cmake_minimum_required(VERSION 3.13)

include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)

project(CO2_Control_System C CXX ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ============================================================
# Pico SDK
# ============================================================

pico_sdk_init()

# ============================================================
# FreeRTOS
# ============================================================

set(FREERTOS_KERNEL_PATH
    ${CMAKE_CURRENT_LIST_DIR}/lib/FreeRTOS-Kernel
)

# ============================================================
# Application
# ============================================================

add_executable(CO2-Control-System
    $ENV{PICO_SDK_PATH}/src/rp2_common/hardware_exception/exception.c

    src/main.cpp
    src/app.cpp
    src/modbus.cpp
    src/sensors.cpp
    src/controller.cpp
    src/ui.cpp
    src/network.cpp
    src/storage.cpp

    # FreeRTOS kernel
    ${FREERTOS_KERNEL_PATH}/tasks.c
    ${FREERTOS_KERNEL_PATH}/queue.c
    ${FREERTOS_KERNEL_PATH}/list.c
    ${FREERTOS_KERNEL_PATH}/timers.c
    ${FREERTOS_KERNEL_PATH}/event_groups.c
    ${FREERTOS_KERNEL_PATH}/stream_buffer.c

    # FreeRTOS heap
    ${FREERTOS_KERNEL_PATH}/portable/MemMang/heap_4.c

    # RP2040 FreeRTOS port
    ${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2040/port.c
)

# ============================================================
# Include directories
# ============================================================

target_include_directories(CO2-Control-System PRIVATE

    ${CMAKE_CURRENT_LIST_DIR}/include

    ${FREERTOS_KERNEL_PATH}/include

    ${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2040/include
    $ENV{PICO_SDK_PATH}/src/rp2_common/hardware_exception/include
    $ENV{PICO_SDK_PATH}/src/rp2_common/hardware_base/include
    $ENV{PICO_SDK_PATH}/src/rp2040/hardware_structs/include

    ${FREERTOS_KERNEL_PATH}/portable/ThirdParty/GCC/RP2040
)

# ============================================================
# Pico SDK libraries
# ============================================================

target_link_libraries(CO2-Control-System

    pico_stdlib

    hardware_uart
    hardware_i2c
    hardware_gpio
    hardware_timer
    hardware_flash
    hardware_sync
    hardware_exception
)

# ============================================================
# USB / UART
# ============================================================

pico_enable_stdio_usb(CO2-Control-System 1)
pico_enable_stdio_uart(CO2-Control-System 0)

# ============================================================
# Extra outputs
# ============================================================

pico_add_extra_outputs(CO2-Control-System)
